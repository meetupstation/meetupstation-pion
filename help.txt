# FFmpeg flags explanation

this piece is adding keyframes every two seconds
`-g 50 -keyint_min 50 -sc_threshold 0`

this other piece is making the negotiation specific for codec - mirroring seems to benefit
`-pix_fmt yuv420p -profile:v baseline -level 5.0`

# to have the loop go through WebRTC
instead use this capture (only the ports changed 4006 -> 4002, 4004 -> 4000)

# screen capture

ffmpeg -f x11grab -video_size 1920x1080 -framerate 25 -i $DISPLAY -threads 4 -c:v libx264 -an -preset ultrafast -tune zerolatency -g 50 -keyint_min 50 -sc_threshold 0 -pix_fmt yuv420p -profile:v baseline -level 5.0 -f rtp rtp://127.0.0.1:4006?pkt_size=1200 -f alsa -ac 2 -sample_rate 48000 -thread_queue_size 1024 -i default -preset ultrafast -vn -c:a libopus -f rtp rtp://127.0.0.1:4004?pkt_size=1200

# webcam capture

ffmpeg -f v4l2 -i /dev/video0 -input_format mjpeg -threads 4 -c:v libx264 -an -preset ultrafast -tune zerolatency -g 50 -keyint_min 50 -sc_threshold 0 -pix_fmt yuv420p -profile:v baseline -level 5.0 -f rtp rtp://127.0.0.1:4006?pkt_size=1200 -f alsa -ac 2 -sample_rate 48000 -i default -vn -c:a libopus -f rtp rtp://127.0.0.1:4004?pkt_size=1200

# playback

ffplay -protocol_whitelist rtp,udp,file \
  -fflags nobuffer \
  -flags low_delay \
  -framedrop \
  -probesize 32 \
  -analyzeduration 0 \
  -sync ext \
  -max_delay 0 \
  -i /path/to/remote.sdp
